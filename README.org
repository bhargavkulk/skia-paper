% Render one state from a comma-list of image filenames.
% Example: \hoarestatefromlist{drawing0.pdf,drawing2.pdf}
\NewDocumentCommand{\hoarestatefromlist}{m}
 {
  \begin{tabular}{c}
    \clist_set:Nn \l_tmpa_clist { #1 }
    \int_zero:N \l_tmpa_int
    \clist_map_inline:Nn \l_tmpa_clist
     {
      \int_incr:N \l_tmpa_int
      \frame{\includegraphics[scale=0.28]{##1}}
      \int_compare:nNnT { \l_tmpa_int } < { \clist_count:N \l_tmpa_clist }
       { \\[0.6em] }
     }
  \end{tabular}
 }

\NewDocumentCommand{\hoarecode}{m}
 {
  \begin{tabular}{T}
  #1
  \end{tabular}

 }
